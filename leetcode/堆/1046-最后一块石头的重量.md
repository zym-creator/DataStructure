class Solution {
public:
    int lastStoneWeight(vector<int>& stones) {
        priority_queue<int> list;
        for (int i=0; i<stones.size(); i++)
            list.push(stones[i]);
        while(list.size()>1){
            int p=list.top();
            list.pop();
            int q=list.top();
            list.pop();
            if(p!=q)
                list.push(p-q);
        }
        return list.empty() ? 0 : list.top();
    }
};
