class Solution {
public:
    int findKthLargest(vector<int>& nums, int k) {
        priority_queue<int,vector<int>,greater<int>> list;
        for(int i=0;i<nums.size();i++){
            list.push(nums[i]);
            if(list.size()>k){
                list.pop();
            }
        }
        return list.top();
    }
};
