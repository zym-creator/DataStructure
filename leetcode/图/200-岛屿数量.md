class Solution {
public:
    int numIslands(vector<vector<char>>& grid) {
        int m = grid.size(),n = grid[0].size();
        int t=0;
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if(grid[i][j]=='1'){
                    fun(grid,i,j);
                    t++;
                }
            }
        } 
        return t;
    }
    bool issafe(vector<vector<char>>& grid, int x, int y){
        return x>=0 && y>=0 && x<grid.size() && y<grid[0].size();
    }
    void fun(vector<vector<char>>& grid, int x, int y){
        if (!issafe(grid,x,y))
            return;
        else if (grid[x][y]=='0')
            return;
        grid[x][y]='0';
        fun(grid,x+1,y);
        fun(grid,x-1,y);
        fun(grid,x,y+1);
        fun(grid,x,y-1);
    }
};
