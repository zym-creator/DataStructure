class Solution {
private:
    vector<int> p;
    vector<int> s;
public:
    int findCircleNum(vector<vector<int>>& isConnected) {
        int n=isConnected.size();
        int t=0;
        for(int i=0; i<n; i++){
            p.push_back(i);
            s.push_back(1);
        }
        for(int i=0; i<n; i++)
            for (int j=i+1; j<n; j++){
                if (isConnected[i][j]==1)
                    unite(i,j);
            }
        for(int i=0; i<n; i++)
            if (p[i]==i) 
                t++;
        return t;
    }
    int find(int x){
        if (p[x]==x)
            return x;
        else
            return p[x]=find(p[x]);
    }
    void unite(int x,int y){
        int rootx = find(x);
        int rooty = find(y);
        if (rootx==rooty)
            return;
        if (s[rootx]>s[rooty]){
            p[rooty]=rootx;
            s[rootx]+=s[rooty];
        }
        else{
            p[rootx]=rooty;
            s[rooty]+=s[rootx];
        }
    }   
};
